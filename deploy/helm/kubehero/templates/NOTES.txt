NEXT STEPS

1. Watch everything come up (the control plane waits for its stores):
   kubectl -n {{ .Release.Namespace }} get pods -w

{{- if and .Values.controlPlane.enabled (not .Values.controlPlane.existingSecret) }}

2. Get your admin token (dashboard sign-in, CLI, API):
   kubectl -n {{ .Release.Namespace }} get secret {{ $fullname }}-control-plane \
     -o jsonpath='{.data.admin-token}' | base64 -d; echo
{{- end }}

{{- if .Values.dashboard.enabled }}

3. Open the dashboard and sign in with that token:
   kubectl -n {{ .Release.Namespace }} port-forward svc/{{ $fullname }}-dashboard {{ .Values.dashboard.service.port }}:{{ .Values.dashboard.service.port }}
   open http://localhost:{{ .Values.dashboard.service.port }}
{{- end }}

4. Use the CLI (binaries on https://github.com/kubehero-io/platform/releases):
   kubectl -n {{ .Release.Namespace }} port-forward svc/{{ $fullname }}-control-plane {{ .Values.controlPlane.service.port }}:{{ .Values.controlPlane.service.port }}
   export KUBEHERO_ENDPOINT=http://localhost:{{ .Values.controlPlane.service.port }} KUBEHERO_TOKEN=<admin token>
   kubehero cost allocation --aggregate namespace --window 7d
   kubehero logs '{namespace="default"} |= "error"' --since 1h
   kubehero ask "what changed in spend this week?"

5. Apply your first guarded policy (samples: https://github.com/kubehero-io/platform/tree/main/config/samples):
   kubectl apply -f https://raw.githubusercontent.com/kubehero-io/platform/main/config/samples/rightsizing-dev.yaml

{{- if and .Values.prometheus.enabled (.Capabilities.APIVersions.Has "monitoring.coreos.com/v1/ServiceMonitor") }}

   Grafana can also query KubeHero logs as a Loki datasource: {{ include "kubehero.controlPlaneURL" . }} (see docs/compatibility).
{{- end }}
